    <div class="use-case-grid">
      <div class="use-case appropriate">
        <h4>Appropriate for Email</h4>
        <ul>
          <li>Research collaboration inquiries</li>
          <li>Project feasibility discussions</li>
          <li>Data analysis consultation requests</li>
          <li>Professional networking</li>
          <li>Speaking or teaching opportunities</li>
        </ul>
      </div>

      <div class="use-case not-appropriate">
        <h4>Better Alternatives</h4>
        <ul>
          <li>Urgent clinical matters (contact relevant clinical team)</li>
          <li>Patient-specific data queries (use proper clinical channels)</li>
          <li>Immediate technical support (use institutional IT)</li>
        </ul>
      </div>

  <div class="email-tips">
    <h3>Tips for Effective Communication</h3>
    <div class="tip-cards">
      <div class="tip-card">
        <h5>Clear Subject Line</h5>
        <p>Include project focus and purpose (e.g., "Collaboration: Lymphoma Survival Analysis")</p>
      </div>

      <div class="tip-card">
        <h5>Brief Project Overview</h5>
        <p>2–3 sentences describing your research question or project goals</p>
      </div>

      <div class="tip-card">
        <h5>Data Availability</h5>
        <p>Mention data types, sizes, and any existing approvals</p>
      </div>

      <div class="tip-card">
        <h5>Timeline & Deliverables</h5>
        <p>Include any specific deadlines or expected outcomes</p>
      </div>
    </div>
