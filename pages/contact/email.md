  <div class="email-tips">
    <h3>Tips for Effective Communication</h3>
    <div class="tip-cards">
      <div class="tip-card">
        <h5>Clear subject line</h5>
        <p>E.g., “Collaboration: Lymphoma Survival Analysis (Q1 timeline)”</p>
      </div>
      <div class="tip-card">
        <h5>Brief overview</h5>
        <p>2–3 sentences on your question or project goals</p>
      </div>
      <div class="tip-card">
        <h5>Data context</h5>
        <p>Type, size, approvals/IRB (if applicable)</p>
      </div>
      <div class="tip-card">
        <h5>Timeline & deliverables</h5>
        <p>Deadlines and the outcome you need</p>
      </div>
